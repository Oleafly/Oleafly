  \epsilon(t) = \alpha e^{-\lambda t} + \beta,

\begin{equation}
  f(x) =
  \begin{cases}
    x^2, & x < 1, \\
    \sqrt{x}, & x \geq 1.
  \end{cases}
\end{equation}

\begin{align}
